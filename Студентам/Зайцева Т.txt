    <script>
        // ===== НАСТРОЙКИ =====
        const ITEMS_PER_PAGE = 3;

        // ===== ДАННЫЕ ДЛЯ ЛЕВОГО СТОЛБЦА (101-110) =====
        const data101 = [
            {
                start: '14.09.2026',
                end: '19.09.2026',
                file: 'Расписание_Дневное/14.09.2026-19.09.2026-101-110-zajceva.pdf'
            },
            {
                start: '07.09.2026',
                end: '12.09.2026',
                file: 'Расписание_Дневное/07.09.2026-12.09.2026-101-110-zajceva.pdf'
            },
            {
                start: '01.09.2026',
                end: '05.09.2026',
                file: 'Расписание_Дневное/01.09.2026-05.09.2026-101-110-zajceva.pdf'
            },
            {
                start: '22.06.2026',
                end: '27.06.2026',
                file: 'Расписание_Дневное/22.06.2026-27.06.2026-101-109-zajceva.pdf'
            }
        ];

        // ===== ДАННЫЕ ДЛЯ ПРАВОГО СТОЛБЦА (ЗАЙЦЕВА 28) =====
        const dataZajceva = [
            {
                start: '14.09.2026',
                end: '19.09.2026',
                file: 'Расписание_Дневное/14.09.2026-19.09.2026-zajceva-28.pdf'
            },
            {
                start: '07.09.2026',
                end: '12.09.2026',
                file: 'Расписание_Дневное/07.09.2026-12.09.2026-zajceva-28.pdf'
            },
            {
                start: '01.09.2026',
                end: '05.09.2026',
                file: 'Расписание_Дневное/01.09.2026-05.09.2026-zajceva-28.pdf'
            }
        ];

        // =============================================
        // ФУНКЦИЯ: РЕНДЕР С АНИМАЦИЕЙ
        // =============================================
        function renderColumn(data, trackId, paginationId) {
            const track = document.getElementById(trackId);
            const pagination = document.getElementById(paginationId);

            const totalPages = Math.ceil(data.length / ITEMS_PER_PAGE);
            let currentPage = 1;
            let isAnimating = false;

            if (data.length === 0) {
                track.innerHTML = '<div class="empty-message">Расписание скоро появится</div>';
                return;
            }

            // Создаём HTML карточек для страницы
            function buildCardsHTML(page) {
                const start = (page - 1) * ITEMS_PER_PAGE;
                const end = start + ITEMS_PER_PAGE;
                const pageItems = data.slice(start, end);

                let html = '';
                pageItems.forEach(item => {
                    html += `
                        <div class="schedule-card">
                            <div class="dates">Расписание занятий<br />с ${item.start} по ${item.end}</div>
                            <a href="${item.file}" class="download-btn" target="_blank">Скачать файл</a>
                        </div>
                    `;
                });
                return html;
            }

            // Первая отрисовка (без анимации)
            function showPageInstant(page) {
                track.innerHTML = buildCardsHTML(page);
                track.classList.remove('slide-out-left', 'slide-out-right', 'slide-in-right', 'slide-in-left');
            }

            // Анимированный переход
            function goToPage(newPage) {
                if (isAnimating || newPage === currentPage) return;
                if (newPage < 1 || newPage > totalPages) return;

                isAnimating = true;
                const direction = newPage > currentPage ? 'forward' : 'backward';
                currentPage = newPage;

                // 1. Старые карточки уезжают
                if (direction === 'forward') {
                    track.classList.add('slide-out-left');
                } else {
                    track.classList.add('slide-out-right');
                }

                // 2. Через 400мс — подменяем содержимое и вводим новые
                setTimeout(() => {
                    // Подменяем содержимое
                    track.innerHTML = buildCardsHTML(currentPage);

                    // Сбрасываем классы
                    track.classList.remove('slide-out-left', 'slide-out-right');

                    // Ставим новые карточки "за экраном"
                    if (direction === 'forward') {
                        track.classList.add('slide-in-right');
                    } else {
                        track.classList.add('slide-in-left');
                    }

                    // Форсируем пересчёт стилей
                    void track.offsetWidth;

                    // 3. Убираем класс — карточки плавно въезжают
                    requestAnimationFrame(() => {
                        track.classList.remove('slide-in-right', 'slide-in-left');
                        setTimeout(() => {
                            isAnimating = false;
                        }, 500);
                    });

                    // Обновляем активную кнопку пагинации
                    updatePagination();
                }, 400);
            }

            // Обновление пагинации
            function updatePagination() {
                if (totalPages > 1) {
                    let pagHtml = '';
                    for (let i = 1; i <= totalPages; i++) {
                        pagHtml += `<button class="${i === currentPage ? 'active' : ''}" data-page="${i}">${i}</button>`;
                    }
                    pagination.innerHTML = pagHtml;

                    pagination.querySelectorAll('button').forEach(btn => {
                        btn.addEventListener('click', function() {
                            goToPage(parseInt(this.dataset.page));
                        });
                    });
                } else {
                    pagination.innerHTML = '';
                }
            }

            // Инициализация
            showPageInstant(1);
            updatePagination();
        }

        // =============================================
        // ЗАПУСК
        // =============================================
        renderColumn(data101, 'track-101-110', 'pagination-101-110');
        renderColumn(dataZajceva, 'track-zajceva-28', 'pagination-zajceva-28');
    </script>
