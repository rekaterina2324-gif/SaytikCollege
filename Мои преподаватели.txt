    <script>
        (function() {
            // =============================================
            // 1. КАРУСЕЛИ
            // =============================================
            function initCarousel(wrapper) {
                const track = wrapper.querySelector('.carousel-track');
                const cards = track.querySelectorAll('.teacher-card');
                const total = cards.length;

                if (total <= 1) {
                    const btns = wrapper.querySelectorAll('.carousel-btn');
                    btns.forEach(b => b.style.display = 'none');
                    return;
                }

                let currentIndex = 0;
                let isTransitioning = false;

                const prevBtn = wrapper.querySelector('.carousel-btn.prev');
                const nextBtn = wrapper.querySelector('.carousel-btn.next');
                const dotsContainer = wrapper.querySelector('.carousel-dots');

                dotsContainer.innerHTML = '';
                for (let i = 0; i < total; i++) {
                    const dot = document.createElement('button');
                    dot.classList.add('dot');
                    if (i === 0) dot.classList.add('active');
                    dot.dataset.index = i;
                    dot.addEventListener('click', function() {
                        goTo(parseInt(this.dataset.index));
                    });
                    dotsContainer.appendChild(dot);
                }
                const dots = dotsContainer.querySelectorAll('.dot');

                function goTo(index) {
                    if (isTransitioning || index === currentIndex) return;
                    if (index < 0) index = total - 1;
                    if (index >= total) index = 0;
                    isTransitioning = true;
                    currentIndex = index;
                    track.style.transform = `translateX(-${currentIndex * 100}%)`;
                    dots.forEach((d, i) => d.classList.toggle('active', i === currentIndex));
                    setTimeout(() => { isTransitioning = false; }, 500);
                }

                function next() { goTo(currentIndex + 1); }

                function prev() { goTo(currentIndex - 1); }

                prevBtn.addEventListener('click', prev);
                nextBtn.addEventListener('click', next);

                // Свайп
                let startX = 0,
                    startY = 0,
                    isSwiping = false;
                track.addEventListener('touchstart', function(e) {
                    startX = e.touches[0].clientX;
                    startY = e.touches[0].clientY;
                    isSwiping = true;
                }, { passive: true });
                track.addEventListener('touchend', function(e) {
                    if (!isSwiping) return;
                    const endX = e.changedTouches[0].clientX;
                    const endY = e.changedTouches[0].clientY;
                    const diffX = startX - endX;
                    const diffY = startY - endY;
                    if (Math.abs(diffX) > Math.abs(diffY) && Math.abs(diffX) > 30) {
                        if (diffX > 0) next();
                        else prev();
                    }
                    isSwiping = false;
                }, { passive: true });

                // Клавиши
                document.addEventListener('keydown', function(e) {
                    if (!wrapper.contains(document.activeElement) && !wrapper.contains(e.target)) return;
                    if (e.key === 'ArrowRight') next();
                    if (e.key === 'ArrowLeft') prev();
                });

                wrapper._cards = cards;
                wrapper._goTo = goTo;
                wrapper._getCurrentIndex = () => currentIndex;
            }

            document.querySelectorAll('.carousel-wrapper').forEach(initCarousel);

            // =============================================
            // 2. ПОИСК
            // =============================================
            const searchInput = document.getElementById('searchInput');
            const resultsContainer = document.getElementById('searchResults');

            const allCards = document.querySelectorAll('.teacher-card');
            const teachers = [];
            allCards.forEach(card => {
                const name = card.dataset.name || '';
                const commission = card.dataset.commission || '';
                let id = card.id;
                if (!id) {
                    id = 'teacher-' + Math.random().toString(36).substr(2, 6);
                    card.id = id;
                }
                const section = card.closest('.section');
                const sectionId = section ? section.id : '';
                teachers.push({ id, name, commission, sectionId, element: card, section });
            });

            function search(query) {
                const q = query.toLowerCase().trim();
                if (q === '') {
                    resultsContainer.classList.remove('active');
                    allCards.forEach(c => c.classList.remove('highlight'));
                    return [];
                }
                const results = teachers.filter(t => t.name.toLowerCase().includes(q));
                renderResults(results, q);
                allCards.forEach(c => c.classList.remove('highlight'));
                results.forEach(r => r.element.classList.add('highlight'));
                return results;
            }

            function renderResults(results, query) {
                if (results.length === 0) {
                    resultsContainer.innerHTML = `<div class="no-result">👀 Преподаватель не найден</div>`;
                    resultsContainer.classList.add('active');
                    return;
                }
                let html = '';
                results.forEach(r => {
                    const index = r.name.toLowerCase().indexOf(query.toLowerCase());
                    let highlightedName = r.name;
                    if (index !== -1) {
                        const before = r.name.substring(0, index);
                        const match = r.name.substring(index, index + query.length);
                        const after = r.name.substring(index + query.length);
                        highlightedName = before + '<strong>' + match + '</strong>' + after;
                    }
                    html += `
                        <div class="result-item" data-id="${r.id}" data-section="${r.sectionId}">
                            <span class="name">${highlightedName}</span>
                            <span class="commission">${r.commission}</span>
                        </div>
                    `;
                });
                resultsContainer.innerHTML = html;
                resultsContainer.classList.add('active');

                resultsContainer.querySelectorAll('.result-item').forEach(item => {
                    item.addEventListener('click', function() {
                        const id = this.dataset.id;
                        const sectionId = this.dataset.section;
                        const card = document.getElementById(id);
                        if (card) {
                            const wrapper = card.closest('.carousel-wrapper');
                            if (wrapper) {
                                const track = wrapper.querySelector('.carousel-track');
                                const cards = track.querySelectorAll('.teacher-card');
                                let index = 0;
                                cards.forEach((c, i) => { if (c.id === id) index = i; });
                                if (wrapper._goTo) wrapper._goTo(index);
                            }
                            if (sectionId) {
                                const section = document.getElementById(sectionId);
                                if (section) section.scrollIntoView({ behavior: 'smooth', block: 'start' });
                            }
                            allCards.forEach(c => c.classList.remove('highlight'));
                            card.classList.add('highlight');
                            setTimeout(() => card.classList.remove('highlight'), 3000);
                            resultsContainer.classList.remove('active');
                            searchInput.value = '';
                        }
                    });
                });
            }

            searchInput.addEventListener('input', function() { search(this.value); });

            document.addEventListener('click', function(e) {
                const wrapper = document.querySelector('.search-wrapper');
                if (!wrapper.contains(e.target)) {
                    resultsContainer.classList.remove('active');
                    allCards.forEach(c => c.classList.remove('highlight'));
                }
            });

            document.addEventListener('keydown', function(e) {
                if (e.key === 'Escape') {
                    resultsContainer.classList.remove('active');
                    allCards.forEach(c => c.classList.remove('highlight'));
                    searchInput.blur();
                }
            });

            searchInput.addEventListener('blur', function() {
                setTimeout(() => {
                    const active = document.activeElement;
                    if (!resultsContainer.contains(active)) {}
                }, 150);
            });

        })();
    </script>
